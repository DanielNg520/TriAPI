```python
from __future__ import annotations

import httpx

from semai.core.results import Result
from semai.config.schema import Settings

"""Minimal SearXNG-only web search worker, general-purpose and restored from the deleted workers/research.py.
This replacement is SearXNG-only, with no other providers and no quota tracking."""


def _search_searxng(query: str, url: str) -> list[dict[str, str]]:
    with httpx.Client(timeout=10.0) as client:
        resp = client.get(url.rstrip("/") + "/search" if "/search" not in url else url, params={"q": query, "format": "json"})
        resp.raise_for_status()
        data = resp.json()
        results = []
        items = data.get("results") or data.get("hits") or []
        for r in items:
            snippet = r.get("content") or r.get("snippet") or ""
            results.append({
                "title": r.get("title", ""),
                "url": r.get("url", ""),
                "snippet": snippet.strip()
            })
        return results


class WebSearchWorker:
    def __init__(self, settings: Settings, client: object | None = None) -> None:
        self.settings = settings
        self.client = client or httpx

    def available(self) -> bool:
        return bool(getattr(self.settings, "searxng_url", None))

    def __call__(self, intent) -> Result:
        query = getattr(intent, "query", "") or ""

        if not self.available():
            return Result(ok=False, message="SearXNG is not configured (searxng_url unset)")

        try:
            results = _search_searxng(query, self.settings.searxng_url)
        except Exception as e:
            return Result(ok=False, message=f"web search failed: {e}")

        if not results:
            return Result(ok=True, message="No results found.", data={"results": []})

        top_results = results[:5]
        lines = []
        for i, r in enumerate(top_results, 1):
            lines.append(f"{i}. {r['title']}\n   {r['url']}\n   {r['snippet']}")
        message = "\n\n".join(lines)

        return Result(ok=True, message=message, data={"results": results})
```
