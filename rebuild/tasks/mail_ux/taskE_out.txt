@pytest.mark.asyncio
async def test_send_unescapes_html_entities_and_strips_invisible_chars_in_mail_ping(monkeypatch):
    adapter = TelegramAdapter(DummyTelegramConfig(), Store())
    payload = "mail_ping:" + json.dumps({
        "msg_id": "m10",
        "sender": "shop@example.com",
        "subject": 'Delivery update: &quot;2Pcs 1.3&quot; OLED Display\u034f',
        "snippet": "",
    })

    sent = {}

    async def fake_post(url, data=None, **kwargs):
        sent["params"] = json.loads(data["params"])
        return FakeResp(200, '{"ok": true}')

    async def fake_aclose():
        sent["aclosed"] = True

    monkeypatch.setattr(adapter._session, "post", fake_post)
    monkeypatch.setattr(adapter._session, "aclose", fake_aclose)

    await adapter._send(123, payload)

    text = sent["params"]["text"]
    assert '"2Pcs 1.3" OLED Display' in text
    assert "&quot;" not in text
    assert "\u034f" not in text
