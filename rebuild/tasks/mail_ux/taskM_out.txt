from semai.providers.ollama import OllamaProvider
from semai.workers.mail_watcher import MailWatcher


class SummarizeMailWorker:
    """Fetches email metadata and returns a short summary using Ollama."""

    def __init__(self, settings):
        self.settings = settings

    def __call__(self, intent) -> Result:
        token = gmail_access_token()
        url = (
            "https://gmail.googleapis.com/gmail/v1/users/me/messages/"
            f"{intent.msg_id}?format=metadata&metadataHeaders=Subject&metadataHeaders=From"
        )
        headers = {"Authorization": f"Bearer {token}"}
        try:
            resp = httpx.get(url, headers=headers, timeout=10.0)
            resp.raise_for_status()
            data: dict[str, Any] = resp.json()

            sender = MailWatcher._extract_sender(data)
            subject = MailWatcher._extract_subject(data)
            snippet = data.get("snippet", "")

            prompt = (
                f"Summarize the following email in 2-3 sentences.\n\n"
                f"From: {sender}\n"
                f"Subject: {subject}\n"
                f"Snippet: {snippet}\n\n"
                "Summary:"
            )

            provider = OllamaProvider(self.settings.ollama_base_url)
            summary = provider.chat(self.settings.ollama_model, prompt)
            return Result(ok=True, message=summary)
        except Exception as exc:
            return Result(ok=False, message=str(exc))
