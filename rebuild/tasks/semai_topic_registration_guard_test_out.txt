```python
from pathlib import Path
import re

from semai.adapters.telegram import TelegramAdapter


def test_all_deliver_topic_names_are_registered():
    root = Path(__file__).parent.parent
    files = [
        root / "src/semai/adapters/daemon.py",
        root / "src/semai/workers/watcher.py",
    ]

    pattern = r'deliver\(\s*[^,]+,\s*store\s*,\s*"([^"]+)"'
    topic_names = []
    for path in files:
        topic_names.extend(re.findall(pattern, path.read_text(), flags=re.DOTALL))

    assert topic_names, "no deliver() topic-name arguments found"

    missing = set(topic_names) - set(TelegramAdapter._FORUM_TOPICS)
    assert missing == set(), f"deliver() topic names not in _FORUM_TOPICS: {missing}"
```
